--- WebSocket protocol (server receives from browser) ---

All messages are JSON. Server receives:
  {"type": "load_route", "waypoints": [[lat, lng], ...]}
  {"type": "play"}
  {"type": "pause"}
  {"type": "stop"}
  {"type": "set_speed", "value": <float>}   # cruise speed in m/s

--- WebSocket protocol (server sends to browser) ---

Server sends at TICK_HZ (20 Hz):
  {"lat": <float>, "lng": <float>, "heading": <float>, "speed": <float>, "state": <str>}
  (This is TelemetryFrame.to_dict())

@app.websocket("/ws")
async def websocket_endpoint(websocket: WebSocket):
    await websocket.accept()
    engine = SimEngine()
    cruise_speed_mps = 10.0  # default, updated by set_speed messages

    async def receive_loop():
        nonlocal cruise_speed_mps
        try:
            async for text in websocket.iter_text():
                msg = json.loads(text)
                t = msg.get("type")
                if t == "load_route":
                    wps = [tuple(p) for p in msg["waypoints"]]
                    try:
                        engine.load_route(wps)
                    except ValueError:
                        pass  # fewer than 2 waypoints — ignore
                elif t == "play":
                    engine.play()
                elif t == "pause":
                    engine.pause()
                elif t == "stop":
                    engine.stop()
                elif t == "set_speed":
                    cruise_speed_mps = float(msg.get("value", cruise_speed_mps))
        except WebSocketDisconnect:
            pass

    async def tick_loop():
        dt = 1.0 / TICK_HZ
        try:
            while True:
                frame = engine.tick(dt, cruise_speed_mps=cruise_speed_mps)
                await websocket.send_text(json.dumps(frame.to_dict()))
                await asyncio.sleep(dt)
        except (WebSocketDisconnect, Exception):
            pass

    await asyncio.gather(receive_loop(), tick_loop())

--- WebSocket ---
function connectWS():
  ws = new WebSocket('ws://' + location.host + '/ws');
  ws.onmessage = (event) => {
    const frame = JSON.parse(event.data);
    simState = frame.state;
    // Update drone marker position
    const latlng = L.latLng(frame.lat, frame.lng);
    droneMarker.setLatLng(latlng);
    map.panTo(latlng);
    // Update heading rotation on inner div
    setDroneHeading(frame.heading);
    // Update drone opacity
    const opacity = frame.state === 'playing' ? 1.0 : 0.4;
    const inner = droneMarker.getElement()?.querySelector('#drone-icon-inner');
    if (inner) inner.style.opacity = opacity;
    // Sync button states
    updateControlStates();
  };
  ws.onclose = () => setTimeout(connectWS, 2000);  // auto-reconnect

Call connectWS() on DOMContentLoaded.

function updateControlStates():
  const canEdit = simState === 'stopped';
  const canPlay = canEdit && waypoints.length >= 2;
  btn-play.disabled = !canPlay;
  btn-pause.disabled = simState !== 'playing';
  btn-stop.disabled = simState === 'stopped';
  btn-clear.disabled = !canEdit;
  btn-draw.disabled = !canEdit;
  // Lock/unlock waypoint drag and click-to-add during simulation
  if (!canEdit && drawModeActive):
    drawModeActive = false;
    map.off('click', onMapClick);
    updateDrawButtonStyle();
