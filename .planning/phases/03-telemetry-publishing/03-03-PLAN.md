Add onTelemetryFrame function:

  function onTelemetryFrame(frame) {
    if (frame.state === 'stopped') {
      // Clear all sparkline history and reset gauges
      for (const k of Object.keys(history)) history[k] = [];
      for (const k of Object.keys(gaugeEls)) gaugeEls[k].textContent = '0.0';
      return;
    }
    if (frame.state === 'playing') {
      // Push new values and cap rolling window
      const values = {
        heading: frame.heading,
        speed:   frame.speed * 1.94384,  // m/s -> knots for display (matches what zenoh subscriber sees)
        roll:    frame.roll,
        pitch:   frame.pitch,
      };
      for (const key of Object.keys(history)) {
        history[key].push(values[key]);
        if (history[key].length > HISTORY_SAMPLES) history[key].shift();
      }
      // Update gauge text
      gaugeEls.heading.textContent = frame.heading.toFixed(1);
      gaugeEls.speed.textContent   = (frame.speed * 1.94384).toFixed(2);
      gaugeEls.roll.textContent    = frame.roll.toFixed(2);
      gaugeEls.pitch.textContent   = frame.pitch.toFixed(2);
    }
    // Redraw sparklines (both playing and paused — paused keeps last state)
    for (const key of Object.keys(svgEls)) {
      if (history[key].length > 1) {
        sparkline.sparkline(svgEls[key], history[key]);
      }
    }
  }

  document.getElementById('btn-settings-save').addEventListener('click', function() {
    if (!ws || ws.readyState !== WebSocket.OPEN) return;
    const subjectsPayload = {};
    for (const inp of document.querySelectorAll('#subject-rows input')) {
      const subj = inp.dataset.subj;
      const field = inp.dataset.field;
      if (!subjectsPayload[subj]) subjectsPayload[subj] = {};
      subjectsPayload[subj][field] = field === 'rate_hz' ? parseFloat(inp.value) : inp.value;
    }
    ws.send(JSON.stringify({
      type: 'update_settings',
      subjects: subjectsPayload,
      roll_amplitude_deg:     parseFloat(document.getElementById('sl-roll-amp').value),
      roll_period_s:          parseFloat(document.getElementById('sl-roll-period').value),
      pitch_amplitude_deg:    parseFloat(document.getElementById('sl-pitch-amp').value),
      pitch_period_s:         parseFloat(document.getElementById('sl-pitch-period').value),
      location_noise_sigma_m: parseFloat(document.getElementById('sl-loc-noise').value),
    }));
    document.getElementById('settings-overlay').style.display = 'none';
  });
