    ```python
    # ghost_drone/zenoh_bridge/publisher.py
    """
    Zenoh publisher — daemon thread + queue pattern.

    zenoh-python 1.x is sync-only (async support was dropped in v0.6).
    All publishing MUST go through this queue. Never call zenoh from
    an asyncio callback, websocket handler, or subscriber callback.

    Phase 1: Scaffold only. Not wired to simulation yet (Phase 3).
    """
    import queue
    import threading
    import logging
    from typing import Optional
    import zenoh

    logger = logging.getLogger(__name__)


    class ZenohPublisher:
        """
        Daemon thread that owns a zenoh session and drains a publish queue.

        Usage (Phase 3+):
            pub = ZenohPublisher(endpoint="tcp/127.0.0.1:7447")
            pub.start()
            pub.publish(key_expr, payload_bytes)
            pub.stop()
        """

        def __init__(self, endpoint: str, queue_maxsize: int = 256) -> None:
            self._endpoint = endpoint
            self._queue: queue.Queue = queue.Queue(maxsize=queue_maxsize)
            self._thread: Optional[threading.Thread] = None
            self._stop_event = threading.Event()

        def start(self) -> None:
            """Start the daemon publisher thread."""
            self._stop_event.clear()
            self._thread = threading.Thread(
                target=self._run, name="zenoh-publisher", daemon=True
            )
            self._thread.start()
            logger.info("ZenohPublisher started (endpoint=%s)", self._endpoint)

        def stop(self, timeout: float = 2.0) -> None:
            """Signal the publisher to stop and wait for thread exit."""
            self._stop_event.set()
            if self._thread is not None:
                self._thread.join(timeout=timeout)

        def publish(self, key_expr: str, payload: bytes) -> None:
            """
            Enqueue a publish operation. Non-blocking; drops if queue is full.
            Call keelson.enclose() before passing payload here.
            """
            try:
                self._queue.put_nowait((key_expr, payload))
            except queue.Full:
                logger.warning("ZenohPublisher queue full — dropping message on %s", key_expr)

        def _run(self) -> None:
            """Publisher thread body. Opens one zenoh session for its lifetime."""
            conf_str = f'{{"connect": {{"endpoints": ["{self._endpoint}"]}}}}'
            try:
                conf = zenoh.Config.from_json5(conf_str)
                with zenoh.open(conf) as session:
                    logger.info("ZenohPublisher: zenoh session open")
                    while not self._stop_event.is_set():
                        try:
                            key_expr, payload = self._queue.get(timeout=0.1)
                            session.put(key_expr, payload)
                        except queue.Empty:
                            continue
                        except Exception:
                            logger.exception("ZenohPublisher: error during put")
            except Exception:
                logger.exception("ZenohPublisher: failed to open zenoh session")
    ```

    ```python
    """
    Pure unit tests for the keelson enclose/uncover round-trip.
    No live zenoh session required.
    """
    import keelson
    from keelson.payloads.Primitives_pb2 import TimestampedFloat


    def test_enclose_uncover_recovers_original_bytes():
        """
        Wrapping bytes with keelson.enclose() then keelson.uncover()
        must recover the exact original bytes.
        """
        msg = TimestampedFloat()
        msg.value = 42.7
        msg.timestamp.GetCurrentTime()
        payload_bytes = msg.SerializeToString()

        envelope_bytes = keelson.enclose(payload_bytes)

        received_at, enclosed_at, recovered_bytes = keelson.uncover(envelope_bytes)

        assert recovered_bytes == payload_bytes


    def test_enclose_uncover_timestamps_are_ordered():
        """
        received_at must be >= enclosed_at, and enclosed_at must be > 0
        (nanosecond integer timestamps).
        """
        msg = TimestampedFloat()
        msg.value = 0.0
        msg.timestamp.GetCurrentTime()
        payload_bytes = msg.SerializeToString()

        envelope_bytes = keelson.enclose(payload_bytes)
        received_at, enclosed_at, recovered_bytes = keelson.uncover(envelope_bytes)

        assert enclosed_at > 0, "enclosed_at timestamp must be a positive nanosecond integer"
        assert received_at >= enclosed_at, "received_at must not precede enclosed_at"


    def test_enclose_uncover_payload_value_survives():
        """
        The payload value must decode correctly after the enclose/uncover round-trip.
        """
        msg = TimestampedFloat()
        msg.value = 123.456
        msg.timestamp.GetCurrentTime()
        payload_bytes = msg.SerializeToString()

        envelope_bytes = keelson.enclose(payload_bytes)
        _, _, recovered_bytes = keelson.uncover(envelope_bytes)

        recovered_msg = TimestampedFloat()
        recovered_msg.ParseFromString(recovered_bytes)
        assert abs(recovered_msg.value - 123.456) < 1e-6


    def test_known_subject_is_well_known():
        """heading_true_north_deg is used in the Phase 1 test helper — must be well-known."""
        assert keelson.is_subject_well_known("heading_true_north_deg"), (
            "heading_true_north_deg should be a well-known keelson subject. "
            "If this fails, the keelson subjects.yaml may have changed — "
            "update PHASE_SUBJECTS in ghost_drone/startup.py accordingly."
        )


    def test_unknown_subject_is_not_well_known():
        """Arbitrary strings must not pass subject validation."""
        assert not keelson.is_subject_well_known("random_mumbo_jumbo_xyz")
    ```
